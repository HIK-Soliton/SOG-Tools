Install-Module Microsoft.Graph -Scope CurrentUser

Connect-MgGraph -Scopes `
  "User.ReadWrite.All",
  "Group.ReadWrite.All"

# テナント情報
$Domain = "testtenant.onmicrosoft.com"

# ユーザー数
$TotalUsers = 300000

# グループ設定
$GroupCount = 30
$GroupPrefix = "grp"

# 初期パスワード
$InitialPassword = "DUMMY_PASSWORD" # 任意のパスワードを指定

# ログ
$LogFile = ".\create_users.log"

$Groups = @{}

for ($i = 1; $i -le $GroupCount; $i++) {
    $GroupName = "{0}{1:D3}" -f $GroupPrefix, $i

    $ExistingGroup = Get-MgGroup -Filter "displayName eq '$GroupName'" -ConsistencyLevel eventual

    if (-not $ExistingGroup) {
        Write-Host "Creating group $GroupName"
        $NewGroup = New-MgGroup `
            -DisplayName $GroupName `
            -MailEnabled:$false `
            -MailNickname $GroupName `
            -SecurityEnabled:$true
        $Groups[$GroupName] = $NewGroup.Id
    }
    else {
        $Groups[$GroupName] = $ExistingGroup[0].Id
    }
}

# ユーザー作成

for ($i = 1; $i -le $TotalUsers; $i++) {

    $UserName = "testuser{0:D6}" -f $i
    $Upn = "$UserName@$Domain"

    # グループ決定（ラウンドロビン）
    $GroupIndex = (($i - 1) % $GroupCount) + 1
    $GroupName = "{0}{1:D3}" -f $GroupPrefix, $GroupIndex
    $GroupId = $Groups[$GroupName]

    try {
        $User = New-MgUser `
            -AccountEnabled:$true `
            -DisplayName $UserName `
            -MailNickname $UserName `
            -UserPrincipalName $Upn `
            -PasswordProfile @{
                ForceChangePasswordNextSignIn = $false
                Password = $InitialPassword
            }

        # グループに追加
        New-MgGroupMember `
            -GroupId $GroupId `
            -DirectoryObjectId $User.Id

        "$Upn created and added to $GroupName" | Out-File $LogFile -Append
    }
    catch {
        "ERROR $Upn : $_" | Out-File $LogFile -Append
    }

    # 軽いスロットリング対策
    if ($i % 100 -eq 0) {
        Start-Sleep -Milliseconds 500
    }
}
