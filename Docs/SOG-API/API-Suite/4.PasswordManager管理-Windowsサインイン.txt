# Windowsサインイン設定インポート
$file = "C:\Users\hiki.FOURSEASONS\Downloads\windowsSignin.csv"
$tmpfile = "C:\Users\hiki.FOURSEASONS\Downloads\temp.csv"
$filename = (Get-Item $file).Name
$fileBin = [System.IO.File]::ReadAllText($file)

(
  "--$boundary",
  "Content-Disposition: form-data; name=`"type`"$LF",
  "2",
  "--$boundary",
  "Content-Disposition: form-data; name=`"file`"; filename=`"$filename`"",
  "Content-Type: text/csv$LF",
    $fileBin,
    "--$boundary--$LF"
) -join $LF | Set-Content $tmpfile -Encoding UTF8

$uri = $baseUri + "/icon/services/seap/api/settings/password-manager/desktop-sso/import"
$contentType = "multipart/form-data; boundary=$boundary"
$method = "POST"

$result = Invoke-Restmethod -Uri $uri `
    -ContentType $contentType `
    -Method $method `
    -Headers  @{ Authorization = "Bearer $apiKey"} `
    -inFile $tmpfile
$result
