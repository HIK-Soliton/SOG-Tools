# 利用者インポート
$file = "C:\Users\hiki.FOURSEASONS\Downloads\OneGateUser.csv"
$tmpfile = "./" + (Get-Date -format "yyyyMMddHHmmss") + ".csv"
$fileBin = [System.IO.File]::ReadAllText($file)
$boundary = [System.Guid]::NewGuid().ToString()
$LF = "`r`n"

(
  "--$boundary",
  "Content-Disposition: form-data; name=`"encodingId`"$LF",
  "30",
  "--$boundary",
  "Content-Disposition: form-data; name=`"file`"; filename=`"OneGateUser.csv`"",
  "Content-Type: application/octet-stream$LF",
    $fileBin,
    "--$boundary--$LF"
) -join $LF | Set-Content $tmpfile -Encoding UTF8

$uri = $baseUri + "/icon/services/seap/api/employee/import"
$contentType = "multipart/form-data; boundary=$boundary"
$method = "POST"

$result = Invoke-Restmethod -Uri $uri `
    -ContentType $contentType `
    -Method $method `
    -Headers  @{ Authorization = "Bearer $apiKey"} `
    -inFile $tmpfile
